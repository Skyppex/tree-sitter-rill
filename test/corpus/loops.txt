For loops and ranges
================================================================================

rill f<N>(xs: [Sample; N]) Sample {
  state y = 0.0
  for i in 0..N {
    y += xs[i]
  }
  for j in 1..=4 { y = y * 0.5 }
  for x in xs { y += x }
  return y
}
