Invoking, triggering and halting
================================================================================

rill main() Sample {
  on start { current = invoke riff(loop: true) }
  on pad(hit) {
    invoke riff
    invoke 3 riff(tempo: 90bpm * hit.velocity)
    trigger 2 current riff
    trigger note.instance riff
    halt riff
    halt current riff
    invoke lead(pitch: C4, velocity: 1)
  }
  return 0
}
